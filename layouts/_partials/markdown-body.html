{{- /* Schoont RawContent op en herschrijft interne links naar hun .md-variant. Aanroepen met de oorsprongpagina als context (niet met een string): .GetPage resolvet relatieve paden t.o.v. die pagina. Wie de tabellen ongemoeid wil laten, roept aan met (dict "page" . "tabellen" true). Met "html" true wijzen interne links naar de absolute URL van de HTML-pagina, voor losse documenten zoals de .odt. */ -}} {{- $p := . -}} {{- $tabellen := false -}} {{- $html := false -}} {{- if reflect.IsMap . -}} {{- $p = .page -}} {{- $tabellen = .tabellen -}} {{- $html = .html -}} {{- end -}} {{- $content := $p.RawContent -}} {{- /* GFM kent geen tabel zonder koprij. In de gepubliceerde .md wordt table-without-header daarom een lijst; pandoc krijgt een lege koprij, want zonder koprij ziet die geen tabel. Aan die lege koprij herkent scripts/downloads/render-downloads.js de tabel weer. Moet vóór het strippen van de shortcodetags. */ -}} {{- range findRE `\{\{<\s*table-without-header\s*>\}\}[\s\S]*?\{\{<\s*/\s*table-without-header\s*>\}\}` $content -}} {{- $rijen := slice -}} {{- $koprij := "" -}} {{- range split . "\n" -}} {{- $regel := trim . " " -}} {{- if hasPrefix $regel "|" -}} {{- if findRE `^\|[\s:|-]+\|$` $regel -}} {{- $koprij = index (last 1 $rijen) 0 -}} {{- else -}} {{- $rijen = $rijen | append $regel -}} {{- end -}} {{- end -}} {{- end -}} {{- if $koprij }}{{ $rijen = after 1 $rijen }}{{ end -}} {{- $regels := slice -}} {{- if $tabellen -}} {{- $leeg := slice -}} {{- $streep := slice -}} {{- range seq (len (split (trim (index $rijen 0) "|") "|")) -}} {{- $leeg = $leeg | append " " -}} {{- $streep = $streep | append " --- " -}} {{- end -}} {{- $regels = $regels | append (printf "|%s|" (delimit $leeg "|")) (printf "|%s|" (delimit $streep "|")) -}} {{- range $rijen }}{{ $regels = $regels | append . }}{{ end -}} {{- else -}} {{- range $rijen -}} {{- $cellen := apply (split (trim . "|") "|") "trim" "." " " -}} {{- $regels = $regels | append (printf "- **%s:** %s" (index $cellen 0) (delimit (after 1 $cellen) ", ")) -}} {{- end -}} {{- end -}} {{- $content = replace $content . (delimit $regels "\n") -}} {{- end -}} {{- /* Moet vóór het strippen hieronder, anders verdwijnt de emoji mee. */ -}} {{- /* Een getekend diagram staat als shortcode in de tekst; in de download hoort de plaat zelf. De shortcode bewaart hem als /images/diagrammen/.svg. Net als bij links wijst de uitvoer voor buiten de site naar de volledige URL. */ -}} {{- $diagrammen := cond $html (absURL "images/diagrammen") "/images/diagrammen" -}} {{- $content = replaceRE `\{\{<\s*diagram-([a-z0-9-]+)[^>]*>\}\}` (printf "![](%s/$1.svg)" $diagrammen) $content -}} {{- $content = replaceRE `\{\{<\s*emoji\s[^>]*?teken="([^"]+)"[^>]*>\}\}` "$1" $content -}} {{- $content = replaceRE `\{\{<[^>]+>\}\}` "" $content -}} {{- $content = replaceRE `\{\{%[^%]+%\}\}` "" $content -}} {{- $content = replaceRE `>\s*\[!(NOTE|WARNING|TIP|IMPORTANT|INFO|CAUTION)\]` "" $content -}} {{- /* Word en Windows kappen een hyperlink af op 2083 tekens en melden het document daarna als beschadigd. In de uitvoer voor buiten de site wijst zo'n link daarom naar de webpagina, waar hij wel werkt. */ -}} {{- if $html -}} {{- range findRESubmatch `\[([^\]]*)\]\((https?://[^)\s]+)\)` $content -}} {{- $geheel := index . 0 -}} {{- if gt (len (index . 2)) 2083 -}} {{- $content = replace $content $geheel (printf "%s (deze URL is te lang voor dit document, maar op [de site van MijnOverheid Zakelijk](%s) vind je die wel)" (index . 1) $p.Permalink) -}} {{- end -}} {{- end -}} {{- end -}} {{- /* findRESubmatch-groepen: 0=geheel, 1='!', 2=tekst, 3=doel */ -}} {{- range findRESubmatch `(!?)\[([^\]]*)\]\(([^)\s]+)\)` $content -}} {{- $whole := index . 0 -}} {{- $text := index . 2 -}} {{- $dest := index . 3 -}} {{- if eq (index . 1) "!" }}{{ continue }}{{ end -}} {{- if hasPrefix $dest "<" }}{{ continue }}{{ end -}} {{- $u := urls.Parse $dest -}} {{- if or $u.IsAbs (hasPrefix $dest "#") (hasPrefix $dest "//") (hasPrefix $dest "mailto:") (hasPrefix $dest "tel:") (not $u.Path) }}{{ continue }}{{ end -}} {{- $path := strings.TrimPrefix "./" $u.Path -}} {{- $new := "" -}} {{- $target := $p.GetPage $path -}} {{- with $target -}} {{- $isMd := false -}} {{- with .File }}{{ if eq .Ext "md" }}{{ $isMd = true }}{{ end }}{{ end -}} {{- $mdOut := .OutputFormats.Get "markdown" -}} {{- if $html -}} {{- $new = .Permalink -}} {{- else if and $isMd $mdOut -}} {{- $new = $mdOut.RelPermalink -}} {{- else -}} {{- $new = .RelPermalink -}} {{- end -}} {{- else -}} {{- with or ($p.Resources.Get $path) (resources.Get $path) -}} {{- $new = cond $html .Permalink .RelPermalink -}} {{- end -}} {{- end -}} {{- /* Bijvoorbeeld een alias: GetPage vindt die niet, de site stuurt wel door. */ -}} {{- if and $html (not $new) (hasPrefix $path "/") }}{{ $new = absURL $path }}{{ end -}} {{- if $new -}} {{- with $u.RawQuery }}{{ $new = printf "%s?%s" $new . }}{{ end -}} {{- with $u.Fragment }}{{ $new = printf "%s#%s" $new . }}{{ end -}} {{- $content = replace $content $whole (printf "[%s](%s)" $text $new) -}} {{- end -}} {{- end -}} {{- $content = replaceRE `\n{3,}` "\n\n" $content -}} {{- return ($content | strings.TrimSpace | safeHTML) -}}
