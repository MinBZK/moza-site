{{- /* De roadmap uit roadmap.yaml voor de tekstuitvoer (markdown-body). Een tijdlijn (van/tot) wordt een lijst per onderdeel; de oude opzet met kolommen een GFM-tabel, met een cel per kwartaal: Markdown kent geen colspan. */ -}} {{- $bron := .page.Resources.GetMatch .naam -}} {{- with $bron -}}{{- else -}}{{- errorf "Roadmap: %q niet gevonden bij %s" $.naam $.page.File.Path -}}{{- end -}} {{- $data := $bron | transform.Unmarshal -}} {{- $uit := "" -}} {{- if isset $data "van" -}} {{- $r := partial "roadmap-rijen.html" $data -}} {{- $regels := slice -}} {{- range $type := slice "verkenning" "realisatie" "doorbouwen" -}} {{- $regels = $regels | append (printf "- %s: %s" (title $type) (index $data.legenda $type)) -}} {{- end -}} {{- range $groep := $r.groepen -}} {{- $regels = $regels | append "" (printf "**%s**" $groep.naam) "" -}} {{- range $rij := $groep.rijen -}} {{- $delen := slice -}} {{- range $rij.balken }}{{ $delen = $delen | append (printf "%s %s" .type .periode) }}{{ end -}} {{- $naam := $rij.naam }}{{ with $rij.pagina }}{{ $naam = printf "[%s](%s)" $naam . }}{{ end }}{{ with $rij.voluit }}{{ $naam = printf "%s (%s)" $naam . }}{{ end -}} {{- with $rij.wie }}{{ $naam = printf "%s, %s" $naam . }}{{ end -}} {{- $regels = $regels | append (printf "- %s: %s" $naam (delimit $delen "; ")) -}} {{- end -}} {{- end -}} {{- $uit = printf "%s\n\n%s:\n\n%s" $data.intro $data.caption (delimit $regels "\n") -}} {{- else -}} {{- $kop := slice "Onderdeel" -}} {{- $streep := slice "---" -}} {{- range $data.kolommen -}} {{- $kop = $kop | append (cond (isset . "fase_label") (printf "%s (%s)" .label .fase_label) .label) -}} {{- $streep = $streep | append "---" -}} {{- end -}} {{- $regels := slice (printf "| %s |" (delimit $kop " | ")) (printf "| %s |" (delimit $streep " | ")) -}} {{- $leeg := slice -}} {{- range $data.kolommen }}{{ $leeg = $leeg | append "" }}{{ end -}} {{- range $groep := $data.groepen -}} {{- $groepsregel := slice (printf "**%s**" (replace $groep.naam "|" `\|`)) | append $leeg -}} {{- $regels = $regels | append (printf "| %s |" (delimit $groepsregel " | ")) -}} {{- range $rij := $groep.rijen -}} {{- $perKolom := dict -}} {{- range $cel := $rij.cellen -}} {{- if ne $cel.type "leeg" -}} {{- $tekst := $cel.label | default (title $cel.type) -}} {{- with $cel.detail }}{{ $tekst = printf "%s: %s" $tekst . }}{{ end -}} {{- range $cel.kolommen }}{{ $perKolom = merge $perKolom (dict . $tekst) }}{{ end -}} {{- end -}} {{- end -}} {{- $cellen := slice $rij.naam -}} {{- range $data.kolommen }}{{ $cellen = $cellen | append (index $perKolom .kolom | default "") }}{{ end -}} {{- $cellen = apply $cellen "replace" "." "|" `\|` -}} {{- $regels = $regels | append (printf "| %s |" (delimit $cellen " | ")) -}} {{- end -}} {{- end -}} {{- $uit = printf "%s\n\n%s:\n\n%s" $data.intro $data.caption (delimit $regels "\n") -}} {{- end -}} {{- return $uit -}}
