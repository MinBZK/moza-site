      \s*)` `${1}${2}

      ` `

{{- /* Elke tabelwikkel een eigen naam. De nummering gebeurt hier en niet in de
       render hook of de shortcode, omdat die twee niet in documentvolgorde
       draaien: de shortcode is aan de beurt nadat Goldmark klaar is. */ -}}
{{- $stukken := split $inhoud `aria-label="Tabel"` -}}
{{- if gt (len $stukken) 1 -}}
  {{- $genummerd := index $stukken 0 -}}
  {{- range $i, $stuk := after 1 $stukken -}}
    {{- $genummerd = printf `%saria-label="Tabel %d"%s` $genummerd (add $i 1) $stuk -}}
  {{- end -}}
  {{- $inhoud = $genummerd -}}
{{- end -}}
