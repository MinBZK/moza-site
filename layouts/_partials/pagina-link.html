{{- /* De link naar een pagina van deze site, eventueel met een anker: partial "pagina-link.html" (dict "pad" "/contact#blijf-op-de-hoogte" "plek" .Position "bron" .Page) Bestaat de pagina of het anker niet meer, dan breekt de build af. Zo wijst een kaart nooit stil naar een pagina die een andere naam kreeg. Het pad mag ook het adres van een pagina zijn, zoals /weekly//. Een bestand (een pad met een extensie) of een andere host gaat ongecontroleerd door; dat doet htmltest. Een anker telt als het als id in de inhoud van de pagina staat. Een id dat alleen de layout van die pagina maakt, kent de build niet. */ -}} {{- $plek := .plek -}} {{- $pad := .pad -}} {{- $u := urls.Parse $pad -}} {{- $href := "" -}} {{- if or $u.Host (path.Ext $u.Path) -}} {{- $href = $pad -}} {{- else -}} {{- $pagina := site.GetPage $u.Path -}} {{- if not $pagina -}} {{- with where site.Pages "RelPermalink" (printf "%s/" (strings.TrimSuffix "/" $u.Path)) }}{{ $pagina = index . 0 }}{{ end -}} {{- end -}} {{- with $pagina -}} {{- $href = .RelPermalink -}} {{- with $u.RawQuery }}{{ $href = printf "%s?%s" $href . }}{{ end -}} {{- with $u.Fragment -}} {{- $anker := . -}} {{- $bekend := false -}} {{- /* Fragments kent alleen Markdown-koppen, en is leeg bij HTML-content. Zoek dus ook in de inhoud, maar niet in die van de bron: die wordt op dat moment gebouwd, en de build zou op zichzelf wachten. */ -}} {{- with $pagina.Fragments }}{{ $bekend = .Identifiers.Contains $anker }}{{ end -}} {{- $zelf := and $.bron ($pagina.Eq $.bron) -}} {{- if not (or $bekend $zelf) -}} {{- $bekend = strings.Contains (partial "koppen-ids.html" $pagina.Content) (printf ` id="%s"` $anker) -}} {{- end -}} {{- if not $bekend -}} {{- errorf "%s: anker #%s staat niet op %s" $plek $anker $u.Path -}} {{- end -}} {{- $href = printf "%s#%s" $href $anker -}} {{- end -}} {{- else -}} {{- errorf "%s: pagina %q bestaat niet" $plek $pad -}} {{- end -}} {{- end -}} {{- return $href -}}
